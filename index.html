Número Aleatório

import random

# Gera um número aleatório entre 1 e 10
numero_secreto = random.randint(1, 10)

# Pede para o usuário tentar adivinhar
input("Adivinhe o número entre 1 e 10: ")

# Revela o número secreto
print("O número secreto é:", numero_secreto)
